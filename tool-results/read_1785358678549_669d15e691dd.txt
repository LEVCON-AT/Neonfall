  1079→  function stopCurSource() {
  1080→    if (nextTimer) { clearTimeout(nextTimer); nextTimer = null; }

  1087→  // Play track i immediately, fading in over fadeInDur. Schedules the next track
  1088→  // to crossfade in FADE_DUR before this track ends.
  1089→  async function playTrack(i: number, fadeInDur: number) {
  1090→    if (!mCtx || !mVol) return;
  1091→    const buf = await loadTrack(i);
  1092→    if (!buf || !mCtx) return;
  1093→    // If we switched tracks while loading, abort.
  1094→    if (curIdx !== i) return;
  1095→    stopCurSource();
  1096→    curSrc = mCtx.createBufferSource();
  1097→    curSrc.buffer = buf;
  1098→    curSrc.loop = false;
  1099→    const t0 = mCtx.currentTime;
  1100→    const srcGain = mCtx.createGain();
  1101→    srcGain.gain.setValueAtTime(0.0001, t0);
  1102→    srcGain.gain.exponentialRampToValueAtTime(MUSIC_VOL, t0 + fadeInDur);
  1103→    curSrc.connect(srcGain); srcGain.connect(mVol);
  1104→    curSrc.start(t0);
  1105→    // schedule crossfade to next track
  1106→    const dur = buf.duration;
  1107→    const crossStart = Math.max(0, dur - FADE_DUR);
  1108→    if (nextTimer) clearTimeout(nextTimer);
  1109→    nextTimer = setTimeout(() => {
  1110→      // fade out current, start next
  1111→      if (curSrc && mCtx) {
  1112→        const tt = mCtx.currentTime;
  1113→        try {
  1114→          srcGain.gain.cancelScheduledValues(tt);
  1115→          srcGain.gain.setValueAtTime(srcGain.gain.value, tt);
  1116→          srcGain.gain.exponentialRampToValueAtTime(0.0001, tt + FADE_DUR);
  1117→        } catch {}
  1118→      }
  1119→      curIdx = (curIdx + 1) % TRACKS.length;
  1120→      // give the fade-out a moment, then start next (overlapping)
  1121→      setTimeout(() => { playTrack(curIdx, FADE_DUR); updateMusicBar(); }, 50);
  1122→      // stop the old source after the fade completes
  1123→      setTimeout(() => {
  1124→        if (curSrc && curSrc.buffer === buf) {
  1125→          try { curSrc.stop(); } catch {}
  1126→        }
  1127→      }, FADE_DUR * 1000 + 100);
  1128→    }, crossStart * 1000);
  1129→    updateMusicBar();
  1130→  }
