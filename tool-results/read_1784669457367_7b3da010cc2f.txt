   773→    function collide(board, piece) {
   774→        const m = piece.matrix;
   775→        const o = piece.pos;
   776→        for (let y = 0; y < m.length; y++) {
   777→            for (let x = 0; x < m[y].length; x++) {
   778→                if (m[y][x] === 0) continue;
   779→                const boardX = x + o.x;
   780→                const boardY = y + o.y;
   781→                if (boardX < 0 || boardX >= COLS || boardY >= ROWS) return true;
   782→                if (boardY >= 0 && board[boardY][boardX] !== 0) return true;
   783→            }
   784→        }
   785→        return false;
   786→    }

   788→    function merge(board, piece) {
   789→        piece.matrix.forEach((row, y) => {
   790→            row.forEach((value, x) => {
   791→                if (value !== 0) {
   792→                    const by = y + piece.pos.y;
   793→                    const bx = x + piece.pos.x;
   794→                    if (by >= 0) board[by][bx] = piece.color;
   795→                }
   796→            });
   797→        });
   798→    }

   800→    function rotateCW(matrix) {
   801→        const rows = matrix.length;
   802→        const cols = matrix[0].length;
   803→        const result = [];
   804→        for (let x = 0; x < cols; x++) {
   805→            const newRow = [];
   806→            for (let y = rows - 1; y >= 0; y--) newRow.push(matrix[y][x]);
   807→            result.push(newRow);
   808→        }
   809→        return result;
   810→    }

   812→    function rotateCCW(matrix) {

   816→    function playerRotate(dir) {
   817→        const oldMatrix = player.matrix;
   818→        const oldPos = { x: player.pos.x, y: player.pos.y };
   819→        player.matrix = dir < 0 ? rotateCCW(player.matrix) : rotateCW(player.matrix);
   820→        let offset = 1;
   821→        while (collide(board, player)) {
   822→            player.pos.x += offset;
   823→            offset = -(offset + (offset > 0 ? 1 : -1));
   824→            if (Math.abs(offset) > player.matrix[0].length + 1) {
   825→                player.matrix = oldMatrix;
   826→                player.pos = oldPos;
   827→                return;
   828→            }
   829→        }

   916→    function shakeScreen(intensity, duration) {
   917→        let start = null;
   918→        function animate(ts) {
   919→            if (!start) start = ts;
   920→            const elapsed = ts - start;

   922→                const progress = 1 - elapsed / duration;
   923→                const dx = (Math.random() * 2 - 1) * intensity * progress;
   924→                const dy = (Math.random() * 2 - 1) * intensity * progress;
   925→                const rot = (Math.random() * 2 - 1) * intensity * 0.05 * progress;

   927→                requestAnimationFrame(animate);
   928→            } else {

  1065→    function lighten(hex, percent) {
  1066→        const num = parseInt(hex.slice(1), 16);
  1067→        let r = (num >> 16) + Math.round(255 * percent / 100);
  1068→        let g = ((num >> 8) & 0xff) + Math.round(255 * percent / 100);
  1069→        let b = (num & 0xff) + Math.round(255 * percent / 100);
  1070→        r = Math.min(255, Math.max(0, r));
  1071→        g = Math.min(255, Math.max(0, g));
  1072→        b = Math.min(255, Math.max(0, b));
  1073→        return `rgb(${r},${g},${b})`;
  1074→    }

  1076→    function roundedRectPath(context, x, y, w, h, r) {
  1077→        context.beginPath();
  1078→        context.moveTo(x + r, y);
  1079→        context.arcTo(x + w, y, x + w, y + h, r);
